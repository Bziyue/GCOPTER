cmake_minimum_required(VERSION 3.16)
project(corridor_parameterization_experiments LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
set(SPLINE_REFERENCE_INCLUDE "" CACHE PATH "Directory containing the reference SplineTrajectory.hpp and SplineConvexHull.hpp")
if(NOT EXISTS "${SPLINE_REFERENCE_INCLUDE}/SplineTrajectory.hpp")
  message(FATAL_ERROR "Set SPLINE_REFERENCE_INCLUDE to the user-specified reference spline directory")
endif()
add_executable(spline_probe spline_probe.cpp)
target_compile_features(spline_probe PRIVATE cxx_std_17)
target_include_directories(spline_probe PRIVATE "${SPLINE_REFERENCE_INCLUDE}")
target_link_libraries(spline_probe PRIVATE Eigen3::Eigen)
target_compile_options(spline_probe PRIVATE -Wall -Wextra -Wpedantic)
