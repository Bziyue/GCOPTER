cmake_minimum_required(VERSION 3.16)
project(native_splines_experiment LANGUAGES CXX)
find_package(Eigen3 REQUIRED)
option(NATIVE_SPLINE_NATIVE_ARCH "Use the local CPU instruction set for all benchmark arms" OFF)
if(NATIVE_SPLINE_NATIVE_ARCH)
  add_compile_options(-march=native)
endif()
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(../.. direct)
add_library(native_splines geometry.cpp cubic_optimizer.cpp cubic_generator.cpp)
target_compile_features(native_splines PUBLIC cxx_std_17)
target_compile_options(native_splines PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(native_splines PUBLIC .)
target_include_directories(native_splines SYSTEM PRIVATE ../../../gcopter/include)
target_link_libraries(native_splines PUBLIC gcopter_direct_corridor)
enable_testing()
add_executable(native_spline_tests tests/math.cpp)
target_link_libraries(native_spline_tests PRIVATE native_splines)
target_compile_options(native_spline_tests PRIVATE -UNDEBUG)
add_test(NAME native_spline_math COMMAND native_spline_tests)
add_executable(native_optimizer_tests tests/optimizer.cpp)
target_link_libraries(native_optimizer_tests PRIVATE native_splines)
target_compile_options(native_optimizer_tests PRIVATE -UNDEBUG)
add_test(NAME native_optimizer_math COMMAND native_optimizer_tests)
add_executable(native_reference_test tests/reference.cpp)
target_link_libraries(native_reference_test PRIVATE native_splines)
target_include_directories(native_reference_test PRIVATE ../../../gcopter/include)
target_compile_options(native_reference_test PRIVATE -UNDEBUG)
add_test(NAME native_minjerk_reference COMMAND native_reference_test)
add_executable(native_scene_tests tests/scenes.cpp)
target_link_libraries(native_scene_tests PRIVATE native_splines)
target_compile_options(native_scene_tests PRIVATE -UNDEBUG)
add_test(NAME native_synthetic_scenes COMMAND native_scene_tests)
add_executable(native_query_tests tests/query.cpp)
target_link_libraries(native_query_tests PRIVATE native_splines)
target_compile_options(native_query_tests PRIVATE -UNDEBUG)
add_test(NAME native_cubic_query COMMAND native_query_tests)
add_executable(native_spline_benchmark benchmark.cpp)
target_include_directories(native_spline_benchmark PRIVATE ../../../gcopter/tools)
target_link_libraries(native_spline_benchmark PRIVATE native_splines)
add_executable(native_volume_refine volume.cpp)
target_include_directories(native_volume_refine PRIVATE ../../../gcopter/tools)
target_link_libraries(native_volume_refine PRIVATE native_splines)
option(NATIVE_SPLINE_TRAJECTORY "Build the common trajectory backend with frozen native geometry replay" OFF)
if(NATIVE_SPLINE_TRAJECTORY)
  find_package(ompl REQUIRED)
  add_executable(native_trajectory_comparison ../../../gcopter/experiments/trajectory/benchmark.cpp)
  target_compile_definitions(native_trajectory_comparison PRIVATE GCOPTER_NATIVE_SPLINE_EXPERIMENT)
  target_include_directories(native_trajectory_comparison PRIVATE ../shape_spline ../../../gcopter/include ../../../gcopter/tools ${OMPL_INCLUDE_DIRS})
  target_link_directories(native_trajectory_comparison PRIVATE ${OMPL_LIBRARY_DIRS})
  target_link_libraries(native_trajectory_comparison PRIVATE native_splines ${OMPL_LIBRARIES})
endif()
