cmake_minimum_required(VERSION 3.16)
project(gcopter_direct_corridor VERSION 0.1.0 LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(gcopter_direct_corridor src/direct_corridor.cpp)
add_library(gcopter::direct_corridor ALIAS gcopter_direct_corridor)
set_target_properties(gcopter_direct_corridor PROPERTIES EXPORT_NAME direct_corridor)
target_compile_features(gcopter_direct_corridor PUBLIC cxx_std_17)
target_include_directories(gcopter_direct_corridor PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(gcopter_direct_corridor PUBLIC Eigen3::Eigen)
target_compile_options(gcopter_direct_corridor PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(direct_corridor_tests tests/test_corridor.cpp)
  target_link_libraries(direct_corridor_tests PRIVATE gcopter_direct_corridor)
  add_test(NAME direct_corridor_tests COMMAND direct_corridor_tests)
endif()
add_executable(direct_corridor_benchmark tools/benchmark.cpp)
target_link_libraries(direct_corridor_benchmark PRIVATE gcopter_direct_corridor)
add_executable(direct_corridor_cli tools/cli.cpp)
target_link_libraries(direct_corridor_cli PRIVATE gcopter_direct_corridor)
option(DIRECT_CORRIDOR_BUILD_MINKOWSKI_EXPERIMENT "Build the experimental planar Minkowski generator" OFF)
if(DIRECT_CORRIDOR_BUILD_MINKOWSKI_EXPERIMENT)
  add_library(corridor_minkowski_experiment STATIC
    experiments/minkowski_generate.cpp experiments/minkowski_evaluate.cpp
    experiments/minkowski_affine.cpp)
  target_include_directories(corridor_minkowski_experiment PUBLIC experiments)
  target_link_libraries(corridor_minkowski_experiment PUBLIC gcopter_direct_corridor)
  target_compile_options(corridor_minkowski_experiment PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(minkowski_tests experiments/minkowski_tests.cpp)
  target_link_libraries(minkowski_tests PRIVATE corridor_minkowski_experiment)
  add_test(NAME minkowski_tests COMMAND minkowski_tests)
  add_executable(minkowski_benchmark experiments/minkowski_benchmark.cpp)
  target_link_libraries(minkowski_benchmark PRIVATE corridor_minkowski_experiment)
  add_executable(minkowski_volume_test experiments/minkowski_volume_test.cpp)
  target_link_libraries(minkowski_volume_test PRIVATE corridor_minkowski_experiment)
  add_test(NAME minkowski_volume_test COMMAND minkowski_volume_test)
  add_executable(corridor_volume_regression ../gcopter/tests/corridor_volume_test.cpp)
  target_link_libraries(corridor_volume_regression PRIVATE gcopter_direct_corridor)
  add_test(NAME corridor_volume_regression COMMAND corridor_volume_regression)
endif()
option(DIRECT_CORRIDOR_BUILD_ROS2 "Build the seed Path / obstacle cloud ROS2 node" OFF)
if(DIRECT_CORRIDOR_BUILD_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(visualization_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  add_executable(direct_corridor_node src/ros2_node.cpp)
  target_link_libraries(direct_corridor_node PRIVATE gcopter_direct_corridor
    rclcpp::rclcpp ${nav_msgs_TARGETS} ${sensor_msgs_TARGETS}
    ${visualization_msgs_TARGETS} ${std_msgs_TARGETS})
  install(TARGETS direct_corridor_node DESTINATION lib/${PROJECT_NAME})
  ament_export_dependencies(Eigen3)
  ament_export_targets(direct_corridorTargets HAS_LIBRARY_TARGET NAMESPACE gcopter::)
endif()
install(TARGETS gcopter_direct_corridor EXPORT direct_corridorTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS direct_corridor_cli direct_corridor_benchmark DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(EXPORT direct_corridorTargets NAMESPACE gcopter:: DESTINATION lib/cmake/${PROJECT_NAME})
if(DIRECT_CORRIDOR_BUILD_ROS2)
  ament_package()
else()
  include(CMakePackageConfigHelpers)
  configure_package_config_file(cmake/config.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/gcopter_direct_corridorConfig.cmake
    INSTALL_DESTINATION lib/cmake/${PROJECT_NAME})
  write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/gcopter_direct_corridorConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/gcopter_direct_corridorConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/gcopter_direct_corridorConfigVersion.cmake DESTINATION lib/cmake/${PROJECT_NAME})
endif()
