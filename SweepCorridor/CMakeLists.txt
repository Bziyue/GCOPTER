cmake_minimum_required(VERSION 3.16)
project(SweepCorridor VERSION 0.1.0 LANGUAGES CXX)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(SweepCorridor INTERFACE)
add_library(SweepCorridor::SweepCorridor ALIAS SweepCorridor)
target_compile_features(SweepCorridor INTERFACE cxx_std_17)
target_include_directories(SweepCorridor INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(SweepCorridor INTERFACE Eigen3::Eigen)
if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  include(CTest)
endif()
if(BUILD_TESTING AND CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  add_executable(sweep_tests tests/test.cpp tests/odr.cpp tests/queries.cpp)
  target_link_libraries(sweep_tests PRIVATE SweepCorridor)
  target_compile_options(sweep_tests PRIVATE -UNDEBUG)
  add_test(NAME sweep_tests COMMAND sweep_tests)
endif()
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/SweepCorridorConfig.cmake
  INSTALL_DESTINATION lib/cmake/SweepCorridor)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/SweepCorridorConfigVersion.cmake
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(TARGETS SweepCorridor EXPORT SweepCorridorTargets)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT SweepCorridorTargets NAMESPACE SweepCorridor:: DESTINATION lib/cmake/SweepCorridor)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/SweepCorridorConfig.cmake
  ${CMAKE_CURRENT_BINARY_DIR}/SweepCorridorConfigVersion.cmake DESTINATION lib/cmake/SweepCorridor)
option(SWEEP_BUILD_BENCHMARKS "Build frozen-input comparison tool" OFF)
if(SWEEP_BUILD_BENCHMARKS)
  add_executable(sweep_frozen benchmarks/frozen.cpp)
  target_link_libraries(sweep_frozen PRIVATE SweepCorridor)
endif()
